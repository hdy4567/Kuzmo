📂 kzm_kernel_store.ts 로직 설계
이 스토어는 모든 레이어의 상태를 감시하고, 특정 레이어가 활성화될 때 다른 레이어의 zIndex와 Transform을 강제로 조정하는 UI 오케스트레이션의 심장 역할을 하게 됩니다.

TypeScript
/**
 * KZM UI Sovereign Layer System (Z-Index & Layout Orchestrator)
 * 1. Z-Index 주권 관리: 모든 레이어의 높이를 중앙 제어
 * 2. 영토 중재(Spatial Arbitration): 하단 네비게이터 vs 보관함 간섭 해결
 */

import { defineStore } from 'pinia'; // 예시로 Pinia 사용, 프로젝트 스택에 따라 변경 가능

interface LayerMetadata {
  id: string;
  baseZIndex: number;
  isActive: boolean;
  isPeekMode: boolean; // 보관함 전용: 하단만 살짝 보이는 상태
}

export const useKernelStore = defineStore('kzm_kernel', {
  state: () => ({
    // 1. 레이어 등록소 (Z-Index 주권 지도)
    layers: {
      'island-navigator': { id: 'island-navigator', baseZIndex: 1000, isActive: true, isPeekMode: false },
      'archive-system': { id: 'archive-system', baseZIndex: 2000, isActive: false, isPeekMode: false },
      'glass-overlay': { id: 'glass-overlay', baseZIndex: 3000, isActive: false, isPeekMode: false }
    } as Record<string, LayerMetadata>,
    
    globalZIndexOffset: 100, // 포커스 시 추가되는 가중치
  }),

  getters: {
    /**
     * 특정 컴포넌트의 실시간 계산된 Z-Index 반환
     */
    getSovereignZIndex: (state) => (id: string) => {
      const layer = state.layers[id];
      if (!layer) return 0;
      return layer.baseZIndex + (layer.isActive ? state.globalZIndexOffset : 0);
    },

    /**
     * [교통 정리 핵심] 네비게이터가 비켜줘야 하는지 판단
     */
    shouldNavigatorRecede: (state) => {
      const archive = state.layers['archive-system'];
      // 보관함이 완전히 열렸거나(isActive), 피크 모드일 때 네비게이터 간섭 발생
      return archive.isActive || archive.isPeekMode;
    }
  },

  actions: {
    /**
     * UI 오케스트레이터: 레이어 상태 변경 및 충돌 해결
     */
    resolveUIConflict(id: string, action: 'OPEN' | 'CLOSE' | 'PEEK') {
      const target = this.layers[id];
      if (!target) return;

      // 1. 상태 업데이트 (주권 행사)
      if (action === 'OPEN') {
        target.isActive = true;
        target.isPeekMode = false;
      } else if (action === 'PEEK') {
        target.isActive = false;
        target.isPeekMode = true;
      } else {
        target.isActive = false;
        target.isPeekMode = false;
      }

      // 2. [교통 경찰 로직] 하단 영토 분쟁 해결 (Spatial Arbitration)
      this.dispatchTrafficControl();
    },

    /**
     * 레이어 간의 물리적 간섭(Y축 점유)을 강제로 조정하는 메서드
     */
    dispatchTrafficControl() {
      const navigator = this.layers['island-navigator'];
      const archive = this.layers['archive-system'];

      if (archive.isPeekMode) {
        // 보관함이 Peek 상태면 네비게이터의 Z-Index를 낮추고 아래로 살짝 밀어냄
        // 이 로직은 실제 CSS 변수(--nav-offset 등)와 연동됩니다.
        document.documentElement.style.setProperty('--kzm-nav-y-offset', '80px');
        document.documentElement.style.setProperty('--kzm-nav-opacity', '0.5');
      } else if (archive.isActive) {
        // 보관함이 완전 활성화되면 네비게이터를 완전히 숨기거나 최하단으로 강등
        document.documentElement.style.setProperty('--kzm-nav-y-offset', '150px');
        document.documentElement.style.setProperty('--kzm-nav-opacity', '0');
      } else {
        // 기본 상태로 복구
        document.documentElement.style.setProperty('--kzm-nav-y-offset', '0px');
        document.documentElement.style.setProperty('--kzm-nav-opacity', '1');
      }
    }
  }
});
🛠 분석 및 보완 포인트
Z-Index 주권의 중앙화: * 기존에는 tokens.css에서 각자 "내가 짱이야!"라며 99999를 써댔다면, 이제는 kzm_kernel_store가 모든 레이어의 baseZIndex를 관리합니다.

컴포넌트는 본인의 Z-Index를 직접 결정하지 않고, 스토어에서 getSovereignZIndex('id')를 구독(Subscribe)해서 받아가기만 하면 됩니다.

Y-축 공간 경쟁 해결 (The Recede Logic):

단순히 '가리는 것'이 문제가 아니라, **'어떻게 비켜줄 것인가'**가 핵심입니다.

dispatchTrafficControl() 메서드가 CSS Variable을 동적으로 주입하여 네비게이터를 아래로 부드럽게 밀어내는(Transform) 연출을 가능하게 합니다.

오케스트레이터의 진화:

이제 boot() 과정에서 이 스토어의 resolveUIConflict를 각 컴포넌트의 생명주기 훅(Mounted/Unmounted)이나 이벤트 리스너에 바인딩해주기만 하면 됩니다.

**"가려지는 컴포넌트가 없게 하겠다"**는 목표, 이 관제탑 로직 하나면 충분히 달성 가능해 보입니다. 이 방향으로 resolveZIndexConflict 상세 구현 들어가 볼까요? 아니면 특정 레이어(예: 팝업 알림 등)의 예외 케이스를 더 추가해 드릴까요?
